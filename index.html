ЗВЁЗДЫ И ЧЁРНАЯ ДЫРА (Используем твой canvas) const canvas = document.getElementById('space-canvas'); const ctx = canvas.getContext('2d'); let W, H; function resize() { W = canvas.width = window.innerWidth; H = canvas.height = window.innerHeight; } window.addEventListener('resize', resize); resize(); const stars = Array.from({ length: 150 }, () => ({ x: Math.random() * W, y: Math.random() * H, r: Math.random() * 1.5 + 0.2, a: Math.random() * 0.6 + 0.3, tw: Math.random() * 0.02 + 0.005, phase: Math.random() * Math.PI * 2 })); let bhAngle = 0; function drawSpace() { ctx.clearRect(0, 0, W, H); // Звёзды stars.forEach(s => { ctx.globalAlpha = Math.max(0.05, Math.min(1, s.a + Math.sin(Date.now() * 0.001 * s.tw + s.phase) * 0.3)); ctx.fillStyle = '#fff'; ctx.beginPath(); ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2); ctx.fill(); }); ctx.globalAlpha = 1; // Чёрная дыра const bhX = W / 2; const bhY = H * 0.4; const bhR = Math.min(W, H) * 0.18; bhAngle += 0.01; for (let i = 0; i < 6; i++) { const rr = bhR * (1.2 + i * 0.15); ctx.beginPath(); ctx.strokeStyle = `rgba(150, 80, 255, ${0.15 - i * 0.02})`; ctx.lineWidth = 2 + i; ctx.ellipse(bhX, bhY, rr, rr * 0.35, bhAngle + i * 0.3, 0, Math.PI * 2); ctx.stroke(); } const glow = ctx.createRadialGradient(bhX, bhY, bhR * 0.8, bhX, bhY, bhR * 2.2); glow.addColorStop(0, 'rgba(120, 60, 255, 0.35)'); glow.addColorStop(1, 'rgba(0, 0, 0, 0)'); ctx.fillStyle = glow; ctx.beginPath(); ctx.arc(bhX, bhY, bhR * 2.2, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#000'; ctx.beginPath(); ctx.arc(bhX, bhY, bhR, 0, Math.PI * 2); ctx.fill(); ctx.strokeStyle = 'rgba(200, 140, 255, 0.9)'; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(bhX, bhY, bhR + 2, 0, Math.PI * 2); ctx.stroke(); requestAnimationFrame(drawSpace); } requestAnimationFrame(drawSpace); // ========================================== // 2. МУЗЫКА И ФИЛЬТР // ========================================== let audio = document.getElementById('bg-music'); let audioCtx, source, filter, isMusicPlaying = false; // Эту функцию вызывает кнопка "Включить звук" в твоем HTML window.toggleMusic = function() { if (!audioCtx) { try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); source = audioCtx.createMediaElementSource(audio); filter = audioCtx.createBiquadFilter(); filter.type = 'lowpass'; filter.frequency.value = 800; source.connect(filter).connect(audioCtx.destination); } catch (e) { console.log("Ошибка аудио:", e); } } if (audioCtx.state === 'suspended') audioCtx.resume(); if (isMusicPlaying) { audio.pause(); document.getElementById('music-toggle').textContent = '🔇 Включить звук'; } else { audio.play().catch(() => {}); document.getElementById('music-toggle').textContent = '🔊 Выключить звук'; } isMusicPlaying = !isMusicPlaying; }; // ========================================== // 3. МЕНЮ, ПЕРЕХОД И ТРЯСКА // ========================================== const menuBtns = document.querySelectorAll('.menu-btn'); const splash = document.getElementById('splash-screen'); const mainMenu = document.getElementById('main-menu'); const gameScreen = document.getElementById('game-screen'); // Добавляем CSS для тряски, если его нет if (!document.getElementById('shake-style')) { const st = document.createElement('style'); st.id = 'shake-style'; st.textContent = `@keyframes shake { 0%{transform:translate(0,0)} 25%{transform:translate(-4px,3px)} 50%{transform:translate(3px,-4px)} 75%{transform:translate(-3px,-2px)} 100%{transform:translate(0,0)} }`; document.head.appendChild(st); } // Обработчик для кнопок меню (Продолжить, Начать новую) menuBtns.forEach(btn => { btn.addEventListener('click', (e) => { // Отменяем стандартный onclick из HTML, если он там есть e.preventDefault(); // Включаем музыку при первом клике if (!isMusicPlaying) toggleMusic(); // Скрываем меню, показываем экран заставки mainMenu.style.display = 'none'; splash.style.display = 'flex'; document.body.style.animation = 'shake 0.4s linear infinite'; // Через 2 секунды убираем заставку и показываем игру setTimeout(() => { document.body.style.animation = ''; splash.style.display = 'none'; // Показываем игровой экран и карту gameScreen.style.display = 'block'; const mapContainer = document.getElementById('map-container'); if (mapContainer) mapContainer.style.display = 'block'; // Показываем джойстик и кнопки управления const controls = document.getElementById('controls'); if (controls) controls.style.display = 'flex'; }, 2000); }); }); // ========================================== // 4. БОЕВАЯ СИСТЕМА (КЛИК ПО КРЫСЕ) // ========================================== const rat = document.getElementById('rat'); const battleScreen = document.getElementById('battle-screen'); const slashEffect = document.getElementById('slash-effect'); const battleButtons = document.querySelectorAll('.battle-btn'); // Клик по крысе запускает бой if (rat) { rat.addEventListener('click', () => { battleScreen.style.display = 'flex'; // Показываем бой document.getElementById('map-container').style.display = 'none'; // Скрываем карту }); } // Эффект "Разреза Нанами" при атаке if (battleButtons.length > 0) { battleButtons.forEach(btn => { // Предполагаем, что первая кнопка — это "Атака" if (btn.textContent.includes('Атака') || btn.textContent.includes('АТАКА')) { btn.addEventListener('click', () => { // Показываем эффект разреза if (slashEffect) { slashEffect.style.display = 'block'; slashEffect.style.opacity = '1'; // Тряска экрана боя battleScreen.style.animation = 'shake 0.2s 2'; setTimeout(() => { slashEffect.style.opacity = '0'; battleScreen.style.animation = ''; setTimeout(() => { slashEffect.style.display = 'none'; }, 300); }, 300); } // Тут можно добавить логику отнимания HP у крысы const enemyHp = document.getElementById('enemy-hp'); if (enemyHp) { // Пример: отнимаем 10 HP (нужно подогнать под твою логику) let currentHp = parseInt(enemyHp.style.width || '100'); enemyHp.style.width = Math.max(0, currentHp - 20) + '%'; } }); } }); }
